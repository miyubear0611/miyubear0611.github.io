// 免费 Cloudflare Worker 1分钟部署代码（已升级 Dexcom G7 官方两步验证协议）：
export default {
  async fetch(request) {
    const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST,GET,OPTIONS", "Access-Control-Allow-Headers": "Content-Type" };
    if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders });
    try {
      const body = request.method === "POST" ? await request.json() : {};
      const hostMap = {
        us: "https://share2.dexcom.com/ShareWebServices/Services",
        ous: "https://shareous1.dexcom.com/ShareWebServices/Services",
        jp: "https://share.dexcom.jp/ShareWebServices/Services"
      };
      const base = hostMap[body.region] || hostMap.ous;
      const appId = "d89443d2-327c-4a6f-89e5-496bbb0317db";
      const headers = { "Content-Type": "application/json", "Accept": "application/json" };

      // Step 1: AuthenticatePublisherAccount -> get accountId
      const authRes = await fetch(`${base}/General/AuthenticatePublisherAccount`, {
        method: "POST", headers,
        body: JSON.stringify({ accountName: body.accountName, password: body.password, applicationId: appId })
      });
      const authText = await authRes.text();
      if (!authRes.ok) return new Response(JSON.stringify({ error: "Step1_AuthFailed", status: authRes.status, detail: authText }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
      const accountId = authText.replace(/"/g, "").trim();

      // Step 2: LoginPublisherAccountById -> get sessionId
      const loginRes = await fetch(`${base}/General/LoginPublisherAccountById`, {
        method: "POST", headers,
        body: JSON.stringify({ accountId, password: body.password, applicationId: appId })
      });
      const loginText = await loginRes.text();
      if (!loginRes.ok) return new Response(JSON.stringify({ error: "Step2_LoginFailed", status: loginRes.status, detail: loginText }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
      const sessionId = loginText.replace(/"/g, "").trim();

      // Step 3: ReadPublisherLatestGlucoseValues
      const bgRes = await fetch(`${base}/Publisher/ReadPublisherLatestGlucoseValues?sessionId=${sessionId}&minutes=1440&maxCount=1`, { method: "POST", headers });
      const arr = await bgRes.json();
      const latest = Array.isArray(arr) && arr[0] ? arr[0] : null;
      return new Response(JSON.stringify({ ok: true, sessionId, count: Array.isArray(arr) ? arr.length : 0, bg: latest ? latest.Value : null, trend: latest ? latest.Trend : null, wt: latest ? latest.WT : null }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
    } catch (err) {
      return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } });
    }
  }
};
