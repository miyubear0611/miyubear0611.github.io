let cachedSession = { sessionId: "", region: "", accountName: "", ts: 0 };

export default {
  async fetch(request) {
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type"
    };
    if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders });

    try {
      let body = {};
      if (request.method === "POST") {
        try { body = await request.json(); } catch (e) { body = {}; }
      }
      const urlParams = new URL(request.url).searchParams;
      const rawAccount = (body.accountName || urlParams.get("u") || "").trim();
      const password = body.password || urlParams.get("p") || "";
      const prefRegion = body.region || urlParams.get("r") || "ous";

      if (!rawAccount || !password) {
        return new Response(JSON.stringify({ ok: true, msg: "Worker 已就绪！请在网页设置中填入该 Worker 网址并点击测试。" }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" }
        });
      }

      const hostMap = {
        ous: "https://shareous1.dexcom.com/ShareWebServices/Services",
        us: "https://share2.dexcom.com/ShareWebServices/Services",
        jp: "https://share.dexcom.jp/ShareWebServices/Services"
      };
      const appIdMap = {
        ous: "d89443d2-327c-4a6f-89e5-496bbb0317db",
        us: "d89443d2-327c-4a6f-89e5-496bbb0317db",
        jp: "d8665ade-9673-4e27-9ff6-92db4ce13d13"
      };
      const order = prefRegion === "us" ? ["us", "ous", "jp"] : (prefRegion === "jp" ? ["jp", "ous", "us"] : ["ous", "us", "jp"]);

      const jsonHeaders = {
        "Content-Type": "application/json",
        "Accept": "application/json",
        "User-Agent": "Dexcom Share/3.0.2.11 CFNetwork/711.2.23 Darwin/14.0.0"
      };
      const emptyPostHeaders = {
        "Accept": "application/json",
        "Content-Length": "0",
        "User-Agent": "Dexcom Share/3.0.2.11 CFNetwork/711.2.23 Darwin/14.0.0"
      };

      async function fetchLatestBg(base, sid) {
        const url = `${base}/Publisher/ReadPublisherLatestGlucoseValues?sessionId=${sid}&minutes=1440&maxCount=2`;
        let res = await fetch(url, { method: "POST", headers: emptyPostHeaders });
        if (!res.ok) {
          res = await fetch(url, {
            method: "POST",
            headers: jsonHeaders,
            body: JSON.stringify({ sessionId: sid, minutes: 1440, maxCount: 2 })
          });
        }
        if (!res.ok) return { ok: false, status: res.status, text: await res.text() };
        const arr = await res.json();
        return { ok: true, arr: Array.isArray(arr) ? arr : [] };
      }

      const incomingSession = body.sessionId || (cachedSession.accountName === rawAccount ? cachedSession.sessionId : "");
      const incomingReg = body.sessionRegion || cachedSession.region || order[0];

      if (incomingSession && hostMap[incomingReg]) {
        const fast = await fetchLatestBg(hostMap[incomingReg], incomingSession);
        if (fast.ok) {
          const latest = fast.arr[0] || null;
          const prev = fast.arr[1] || null;
          return new Response(JSON.stringify({
            ok: true,
            reusedSession: true,
            sessionId: incomingSession,
            regionUsed: incomingReg,
            count: fast.arr.length,
            bg: latest ? latest.Value : null,
            delta: (latest && prev) ? (latest.Value - prev.Value) : null,
            trend: latest ? latest.Trend : null,
            wt: latest ? (latest.WT || latest.ST) : null
          }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
        }
      }

      const candidates = [rawAccount];
      if (rawAccount.includes("@")) candidates.push(rawAccount.split("@")[0]);

      let lastError = "";
      for (const reg of order) {
        const base = hostMap[reg];
        const appId = appIdMap[reg];
        for (const accName of candidates) {
          let sessionId = "";
          const authRes = await fetch(`${base}/General/AuthenticatePublisherAccount`, {
            method: "POST",
            headers: jsonHeaders,
            body: JSON.stringify({ accountName: accName, password, applicationId: appId })
          });
          const authText = (await authRes.text()).trim();
          if (authRes.ok) {
            const accountId = authText.replace(/"/g, "").trim();
            const loginRes = await fetch(`${base}/General/LoginPublisherAccountById`, {
              method: "POST",
              headers: jsonHeaders,
              body: JSON.stringify({ accountId, password, applicationId: appId })
            });
            const loginText = (await loginRes.text()).trim();
            if (loginRes.ok) {
              sessionId = loginText.replace(/"/g, "").trim();
            } else {
              lastError = `[${reg}/${accName}] Step2 HTTP ${loginRes.status}: ${loginText}`;
            }
          } else {
            lastError = `[${reg}/${accName}] Step1 HTTP ${authRes.status}: ${authText}`;
            const legacyRes = await fetch(`${base}/General/LoginPublisherAccountByName`, {
              method: "POST",
              headers: jsonHeaders,
              body: JSON.stringify({ accountName: accName, password, applicationId: appId })
            });
            if (legacyRes.ok) {
              sessionId = (await legacyRes.text()).replace(/"/g, "").trim();
            }
          }

          if (sessionId && sessionId !== "00000000-0000-0000-0000-000000000000") {
            cachedSession = { sessionId, region: reg, accountName: rawAccount, ts: Date.now() };
            const bgData = await fetchLatestBg(base, sessionId);
            if (bgData.ok) {
              const latest = bgData.arr[0] || null;
              const prev = bgData.arr[1] || null;
              return new Response(JSON.stringify({
                ok: true,
                sessionId,
                regionUsed: reg,
                accountUsed: accName,
                count: bgData.arr.length,
                bg: latest ? latest.Value : null,
                delta: (latest && prev) ? (latest.Value - prev.Value) : null,
                trend: latest ? latest.Trend : null,
                wt: latest ? (latest.WT || latest.ST) : null
              }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
            } else {
              lastError = `[${reg}] Step3 HTTP ${bgData.status}: ${bgData.text}`;
            }
          }
        }
      }

      return new Response(JSON.stringify({ ok: false, error: "AuthFailedAllRegions", detail: lastError }), {
        headers: { ...corsHeaders, "Content-Type": "application/json" }
      });
    } catch (err) {
      return new Response(JSON.stringify({ ok: false, error: err.message }), {
        status: 500,
        headers: { ...corsHeaders, "Content-Type": "application/json" }
      });
    }
  }
};
